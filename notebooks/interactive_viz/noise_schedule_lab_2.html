def linear_sched(betamin=0.0001, betamax=0.02,


                 n_steps=1000):


    beta = torch.linspace(betamin, betamax, n_steps)


    return SimpleNamespace(


        a    = 1. - beta,


        abar = (1.-beta).cumprod(dim=0),


        sig  = beta.sqrt())
