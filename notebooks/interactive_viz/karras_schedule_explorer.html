def sigmas_karras(n, sigma_min=0.01, sigma_max=80., rho=7.): ramp = torch.linspace(0, 1, n) # ① n evenly spaced points min_inv_rho = sigma_min ** (1/rho); max_inv_rho = sigma_max ** (1/rho) # ② warp the endpoints interp = max_inv_rho + ramp * (min_inv_rho - max_inv_rho) # ③ straight line in warped space sigmas = interp ** rho # ④ un-warp → curved in σ-space return torch.cat([sigmas, tensor([0.])]) # append exact 0 (clean)

purple = Karras schedule  •  grey dashed = linear spacing  •  dot color runs orange (high σ) → teal (low σ). ρ=7 is the paper's sweet spot; ρ=1 recovers plain linear spacing.
