def noisify(x0, alphabar):


    n = len(x0)                # batch size = 4


    t = torch.randint(0, n_steps, (n,))


    epsilon = torch.randn(x0.shape)


    alphabar_t = alphabar[t].reshape(-1,1,1,1)


    xt = alphabar_t.sqrt()*x0 \


       + (1-alphabar_t).sqrt()*epsilon


    return (xt, t), epsilon
